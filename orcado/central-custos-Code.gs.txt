function prop_(k) { return PropertiesService.getScriptProperties().getProperty(k) || ""; }

function limpa_(s, max) { return String(s == null ? "" : s).trim().slice(0, max || 2000); }

const num_ = v => { const n = parseFloat(String(v == null ? "" : v).replace(",", ".")); return isFinite(n) ? n : 0; };

function planilha_() {
  const pr = PropertiesService.getScriptProperties();
  const id = pr.getProperty("PLANILHA_ID");
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) { } }
  const ss = SpreadsheetApp.create(NOME_PLANILHA);
  pr.setProperty("PLANILHA_ID", ss.getId());
  return ss;
}
function aba_(t) {
  const def = T[t], ss = planilha_();
  let sh = ss.getSheetByName(def.aba);
  if (!sh) {
    sh = ss.insertSheet(def.aba);
    ["Página1", "Sheet1", "Planilha1"].forEach(n => { const p = ss.getSheetByName(n); if (p && ss.getSheets().length > 1) ss.deleteSheet(p); });
    sh.getRange(1, 1, 1, def.tit.length).setValues([def.tit]).setFontWeight("bold").setBackground("#002B49").setFontColor("#FFFFFF").setWrap(true);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, Math.max(sh.getMaxRows(), 2), def.cols.length).setNumberFormat("@");
    if (t === "visitas") { sh.hideColumns(1); sh.hideColumns(def.cols.length); }
    if (t === "notas") { sh.hideColumns(13); sh.hideColumns(15); sh.hideColumns(16); }
  } else if (sh.getLastColumn() < def.cols.length) {   // coluna nova numa aba que já existia
    sh.getRange(1, 1, 1, def.tit.length).setValues([def.tit]).setFontWeight("bold").setBackground("#002B49").setFontColor("#FFFFFF").setWrap(true);
  }
  return sh;
}
function valor_(v) { return v instanceof Date ? Utilities.formatDate(v, TZ, "yyyy-MM-dd") : v; }
function ler_(t) {
  const def = T[t], sh = aba_(t), n = sh.getLastRow();
  if (n < 2) return [];
  return sh.getRange(2, 1, n - 1, def.cols.length).getValues().map((l, i) => {
    const o = { _linha: i + 2 };
    def.cols.forEach((c, k) => o[c] = valor_(l[k]));
    return o;
  }).filter(o => String(o[def.cols[0]]).trim() !== "");
}
function linha_(t, o) { return T[t].cols.map(c => { const x = o[c] == null ? "" : String(o[c]); return /^[=+\-@]/.test(x) ? "'" + x : x; }); }
function bloco_(sh, ini, n, cols) {
  const falta = ini + n - 1 - sh.getMaxRows();
  if (falta > 0) sh.insertRowsAfter(sh.getMaxRows(), falta);
  return sh.getRange(ini, 1, n, cols);
}
function gravar_(t, o, linha) {
  const def = T[t], sh = aba_(t), v = [linha_(t, o)];
  if (linha) sh.getRange(linha, 1, 1, def.cols.length).setNumberFormat("@").setValues(v);
  else bloco_(sh, sh.getLastRow() + 1, 1, def.cols.length).setNumberFormat("@").setValues(v);
}
function reescrever_(t, lista) {
  const def = T[t], sh = aba_(t), n = sh.getLastRow();
  if (n >= 2) sh.getRange(2, 1, n - 1, def.cols.length).clearContent();
  if (lista.length) bloco_(sh, 2, lista.length, def.cols.length).setNumberFormat("@").setValues(lista.map(o => linha_(t, o)));
}
function pasta_(nome) { const it = DriveApp.getFoldersByName(nome); return it.hasNext() ? it.next() : DriveApp.createFolder(nome); }

/* ---------- regras de cálculo (as mesmas da planilha BD - CUSTOS DE INSTALAÇÃO) ---------- */
// horas técnicas e extras pela MÉDIA GERAL do valor hora (mesma base da DIR do comercial), com a média do ANO da visita:
// ano atual = média dos colaboradores ativos; anos anteriores = média informada (parâmetro media_hora_AAAA) ou, sem ela,
// a média dos valores hora lançados nas visitas daquele ano.
let MEDIA_ = undefined, MEDIAS_ANO_ = {};
const H_DIA_ = HORAS_MES / DIAS_MES;
function mediaCadastro_() {
  if (MEDIA_ !== undefined) return MEDIA_;
  let L = [];
  try { L = ler_("func").length ? funcs_().filter(f => f.vHora > 0) : []; } catch (e) { L = []; }
  MEDIA_ = L.length ? { hora: L.reduce((a, f) => a + f.vHora, 0) / L.length, dia: L.reduce((a, f) => a + f.vDia, 0) / L.length } : null;
  return MEDIA_;
}
function mediasInformadas_() {
  const m = {};
  try { ler_("param").forEach(p => { const k = String(p.chave).match(/^media_hora_(\d{4})$/); if (k && num_(p.valor) > 0) m[k[1]] = num_(p.valor); }); } catch (e) { }
  return m;
}
function media_(v) {
  const ano = String((v && (v.ini || v.fim)) || "").slice(0, 4), atual = Utilities.formatDate(new Date(), TZ, "yyyy");
  if (!/^\d{4}$/.test(ano) || ano >= atual) return mediaCadastro_();
  if (ano in MEDIAS_ANO_) return MEDIAS_ANO_[ano];
  const inf = mediasInformadas_()[ano];
  if (inf) return MEDIAS_ANO_[ano] = { hora: inf, dia: inf * H_DIA_ };
  let soma = 0, n = 0;
  try {
    ler_("visitas").forEach(l => { if (String(l.inicio || "").slice(0, 4) !== ano) return;
      let d = {}; try { d = JSON.parse(String(l.dados || "{}")); } catch (e) { }
      (d.equipe || []).forEach(x => { if (num_(x.vHora) > 0) { soma += num_(x.vHora); n++; } }); });
  } catch (e) { }
  return MEDIAS_ANO_[ano] = n ? { hora: soma / n, dia: soma / n * H_DIA_ } : mediaCadastro_();
}
function totais_(v) {
  const d = v.desl || {}, e = v.est || {}, s = v.serv || {};
  const carro = num_(d.carro) > 150 ? num_(d.carro) : num_(d.carro) * num_(d.diasCarro);  // antigos gravaram o total
  const real = carro + ["aluguel", "uber", "passagem", "pedagio", "bagagem", "combustivel"].reduce((a, k) => a + num_(d[k]), 0)
    + num_(e.hotel) + num_(e.alimentacao) + num_(s.frete) + num_(s.munk)
    + (v.equipe || []).reduce((a, x) => { const m = media_(v) || { dia: num_(x.vDia), hora: num_(x.vHora) };
        return a + m.dia * num_(x.dias) + m.hora * (num_(x.h50) * 1.5 + num_(x.h75) * 1.75); }, 0)
    + (v.adic || []).reduce((a, x) => a + num_(x.valor), 0)
    + (v.notas || []).filter(n => !n.recusada).reduce((a, n) => a + num_(n.valor), 0);
  const o = v.orc;
  const orc = o ? num_(o.desl) + (num_(o.est) || num_(o.hotel) + num_(o.alim)) + num_(o.ht) + num_(o.he) + num_(o.adic) + num_(o.frete) + num_(o.munk) : 0;
  return { orc: r2_(orc), real: r2_(real), saldo: orc ? r2_(orc - real) : "" };
}

  try { v = JSON.parse(String(l.dados || "{}")); } catch (e) { }

  return ler_("func").filter(f => !/^n/i.test(String(f.ativo || ""))).map(f => {
    const tot = num_(f.salario) + num_(f.beneficios) + num_(f.impostos);
    return { nome: String(f.nome).trim(), agenda: String(f.nome_agenda || "").trim(), vHora: r2_(tot / HORAS_MES), vDia: r2_(tot / DIAS_MES) };

}

// notas fiscais lançadas em campo, agrupadas por visita (sem a foto: só a miniatura)

function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p.acao === "ping") return json_({ ok: true, app: "Custos de Campo ECTAS" });
  return json_({ ok: false, erro: "use o Painel do Supervisor" });

function doPost(e) {
  let b;
  try { b = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, erro: "formato inválido" }); }
  if (b && b.campo) return json_(acaoCampo_(b));
  if (!b || !b.painel) return json_({ ok: false, erro: "formato inválido" });
  return json_(acaoSupervisor_(b));
}

// senha do painel = senha do Painel do Diário (conferida na central do Diário, com cache de 10 minutos)
function senhaDiario_(senha) {
  if (!senha) return { ok: false, erro: "senha incorreta" };
  const cache = CacheService.getScriptCache();
  const k = "sd_" + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, "cst|" + senha, Utilities.Charset.UTF_8));
  if (cache.get(k)) return { ok: true };
  try {
    const r = UrlFetchApp.fetch(DIARIO_URL + "?acao=obrasTodas&senha=" + encodeURIComponent(senha) + "&_=" + Date.now(), { muteHttpExceptions: true });
    const j = JSON.parse(r.getContentText());
    if (j.ok) { cache.put(k, "1", 600); return { ok: true }; }
    return { ok: false, erro: j.erro === "senha vencida" ? "senha vencida" : "senha incorreta" };
  } catch (e) { return { ok: false, erro: "Não consegui falar com a central do Diário de Obra. Tente de novo." }; }
}

function acaoSupervisor_(b) {
  const sd = senhaDiario_(b.senha);
  if (!sd.ok) return sd;
  if (b.acao === "listar") return listar_();
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    switch (b.acao) {
      case "lancSalvar": return lancSalvar_(b.lancamento || {});
      case "lancExcluir": return lancExcluir_(String(b.id || ""));
      case "paramSalvar": return paramSalvar_(b.chave, b.valor);
      case "funcListar": return { ok: true, lista: funcLista_() };
      case "funcSalvar": return funcSalvar_(b.colaborador || {});
      case "backupAgora": garantirGatilhoBackup_(); return backupSemanal();
      case "notaFoto": return notaFoto_(String(b.id || ""), null);
    }
    return { ok: false, erro: "ação desconhecida" };
  } finally { lock.releaseLock(); }
}

function listar_() {
  try { garantirGatilhoBackup_(); } catch (e) { }
  const todas = ler_("visitas").map(visita_);
  const nv = notasPorVisita_();
  todas.forEach(v => { if (v.status !== "Histórico") v.notas = nv[v.id] || []; });
  const p = params_();
  return { ok: true, planilha: planilha_().getUrl(),
    visitas: todas.filter(v => v.status === "Histórico"),
    lancamentos: todas.filter(v => v.status !== "Histórico"),
    funcionarios: funcs_(), carroDia: num_(p.carro_dia) || 150, medias: mediasInformadas_() };
}

function lancSalvar_(v) {
  const id = limpa_(v.id, 40) || ("L" + Utilities.getUuid().replace(/-/g, "").slice(0, 12));
  if (!limpa_(v.idObra) || !limpa_(v.pedido) || !ehData_(v.ini) || !ehData_(v.fim)) return { ok: false, erro: "Obra, pedido e datas são obrigatórios." };
  const status = v.status === "Aprovado" ? "Aprovado" : v.status === "Em campo" ? "Em campo" : "Enviado";
  const resp = v.resp && limpa_(v.resp.login) ? { login: limpa_(v.resp.login, 60).toLowerCase(), nome: limpa_(v.resp.nome, 100) } : null;
  if (status === "Em campo" && !resp) return { ok: false, erro: "Escolha o responsável em campo antes de abrir a visita para o técnico." };
  const dados = Object.assign({}, v, { id: id, status: status, resp: resp });
  delete dados._k; delete dados._app;
  const atual = ler_("visitas").find(l => String(l.id) === id);
  if (atual && String(atual.status) === "Histórico") return { ok: false, erro: "Visitas do histórico não são alteradas pelo painel." };
  // notas de campo: ficam na aba Notas; do painel só vale a marcação de recusada
  const rec = {}; (v.notas || []).forEach(n => { if (n && n.id) rec[String(n.id)] = !!n.recusada; });
  const sh = aba_("notas");
  ler_("notas").filter(n => String(n.visita) === id && String(n.id) in rec).forEach(n => {
    const r = rec[String(n.id)] ? "Sim" : "";
    if (String(n.recusada || "") !== r) sh.getRange(n._linha, T.notas.cols.indexOf("recusada") + 1).setValue(r);
  });
  delete dados.notas;
  const notas = notasPorVisita_()[id] || [];
  const t = totais_(Object.assign({}, dados, { notas: notas }));
  gravar_("visitas", { id: id, status: status, origem: "Painel", tipo: limpa_(v.tipo, 40), pedido: limpa_(v.pedido, 30), id_obra: limpa_(v.idObra, 30),
    obra: limpa_(v.obra, 150), inicio: v.ini, fim: v.fim, supervisor: limpa_(v.supervisor, 80), orcado: t.orc, realizado: t.real, saldo: t.saldo,
    lancado_em: atual ? atual.lancado_em : agora_(), aprovado_em: status === "Aprovado" ? (atual && atual.aprovado_em || agora_()) : "",
    dados: JSON.stringify(dados) }, atual ? atual._linha : 0);
  return { ok: true, id: id, status: status, notas: notas };
}
function lancExcluir_(id) {
  const atual = ler_("visitas").find(l => String(l.id) === id);
  if (!atual) return { ok: true };
  if (String(atual.status) === "Histórico") return { ok: false, erro: "Visitas do histórico não são excluídas pelo painel." };
  aba_("visitas").deleteRow(atual._linha);
  ler_("notas").filter(n => String(n.visita) === id).reverse().forEach(n => {
    try { if (n.foto_id) DriveApp.getFileById(String(n.foto_id)).setTrashed(true); } catch (e) { }
    aba_("notas").deleteRow(n._linha);
  });
  return { ok: true };
}
function paramSalvar_(chave, valor) {
  const media = /^media_hora_\d{4}$/.test(String(chave || ""));
  if (["carro_dia"].indexOf(chave) < 0 && !media) return { ok: false, erro: "parâmetro desconhecido" };
  const lista = ler_("param"), x = lista.find(p => p.chave === chave);
  if (media) {   // média do valor hora de um ano anterior; vazio/0 = volta a ser calculada pelos lançamentos
    if (!(r2_(valor) > 0)) { if (x) aba_("param").deleteRow(x._linha); return { ok: true }; }
    gravar_("param", { chave: chave, valor: String(r2_(valor)), descricao: "Média do valor hora usada na DIR em " + chave.slice(-4) + " [R$/h]" }, x ? x._linha : 0);
    return { ok: true };
  }
  gravar_("param", { chave: chave, valor: String(r2_(valor)), descricao: x ? x.descricao : "" }, x ? x._linha : 0);
  return { ok: true };
}

// rode pelo editor para refazer orçado, realizado e saldo da aba Visitas com a regra atual (média geral por ano)
function recalcularTotais() {
  const sh = aba_("visitas"), c = T.visitas.cols, nv = notasPorVisita_();
  const linhas = ler_("visitas");
  linhas.forEach(l => {
    const v = visita_(l); v.notas = nv[v.id] || [];
    const t = totais_(v);
    sh.getRange(l._linha, c.indexOf("orcado") + 1, 1, 3).setValues([[String(t.orc), String(t.real), String(t.saldo)]]);
  });
  Logger.log(linhas.length + " visitas recalculadas.");
}

/* ---------- colaboradores: o supervisor inclui, altera salário e desativa pelo painel ---------- */
function funcLista_() {
  return ler_("func").map(f => ({ nome: String(f.nome).trim(), agenda: String(f.nome_agenda || "").trim(), salario: r2_(f.salario), beneficios: r2_(f.beneficios),
    impostos: r2_(f.impostos), ativo: !/^n/i.test(String(f.ativo || "")), alteradoEm: String(f.alterado_em || ""), historico: String(f.historico || "") }))
    .sort((a, b) => a.nome.localeCompare(b.nome, "pt-BR"));
}
function funcSalvar_(c) {
  const nome = limpa_(c.nome, 80).replace(/\s+/g, " "), orig = limpa_(c.original, 80);
  if (!nome) return { ok: false, erro: "Informe o nome do colaborador." };
  const sal = r2_(c.salario), ben = r2_(c.beneficios), imp = r2_(c.impostos);
  if ([sal, ben, imp].some(x => !(x >= 0))) return { ok: false, erro: "Valores inválidos." };
  if (!(sal > 0)) return { ok: false, erro: "Informe o salário." };
  const lista = ler_("func"), chave = x => String(x).trim().toLowerCase();
  const atual = orig ? lista.find(f => chave(f.nome) === chave(orig)) : null;
  if (orig && !atual) return { ok: false, erro: "Colaborador não encontrado. Atualize a página." };
  if (lista.some(f => f !== atual && chave(f.nome) === chave(nome))) return { ok: false, erro: "Já existe um colaborador com esse nome." };
  const quando = agora_(), custo = x => r2_(num_(x.salario) + num_(x.beneficios) + num_(x.impostos));
  let hist = atual ? String(atual.historico || "") : "";
  if (!atual || custo(atual) !== r2_(sal + ben + imp))   // guarda cada mudança de custo
    hist = (hist ? hist + " | " : "") + quando.slice(0, 10) + ": " + [sal, ben, imp].map(x => String(x).replace(".", ",")).join(" + ");
  gravar_("func", { nome: nome, nome_agenda: limpa_(c.agenda, 80), salario: String(sal).replace(".", ","), beneficios: String(ben).replace(".", ","),
    impostos: String(imp).replace(".", ","), ativo: c.ativo === false ? "Não" : "Sim", alterado_em: quando, historico: hist.slice(-2000) }, atual ? atual._linha : 0);
  MEDIA_ = undefined; MEDIAS_ANO_ = {};
  return { ok: true, lista: funcLista_(), funcionarios: funcs_() };
}

/* ---------- app de campo (técnico): só notas fiscais das visitas em que é o responsável ---------- */
// pergunta à central do Diário quem é o dono do token (cache de 10 minutos); null = inválido, false = sem resposta
function usuarioDiario_(tokenDiario) {
  if (!tokenDiario) return null;
  const cache = CacheService.getScriptCache();
  const k = "td_" + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, "cst|" + tokenDiario, Utilities.Charset.UTF_8));
  const c = cache.get(k);
  if (c) return JSON.parse(c);
  try {
    const r = UrlFetchApp.fetch(DIARIO_URL, { method: "post", contentType: "text/plain;charset=utf-8", muteHttpExceptions: true,
      payload: JSON.stringify({ acao: "obras", chave: DIARIO_CHAVE, token: String(tokenDiario) }) });
    const j = JSON.parse(r.getContentText());
    if (j.ok && j.usuario && j.usuario.login) {
      const u = { login: String(j.usuario.login).toLowerCase(), nome: String(j.usuario.nome || j.usuario.login) };
      cache.put(k, JSON.stringify(u), 600); return u;
    }
    if (j.erro === "sessao") return null;
    return false;
  } catch (e) { return false; }
}
function acaoCampo_(b) {
  const u = usuarioDiario_(b.tokenDiario);
  if (u === false) return { ok: false, erro: "Não consegui falar com a central do Diário de Obra. Tente de novo." };
  if (!u) return { ok: false, erro: "sessao" };
  if (b.acao === "campoListar") return campoListar_(u);
  if (b.acao === "notaFoto") return notaFoto_(String(b.id || ""), u);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    if (b.acao === "notaSalvar") return notaSalvar_(u, String(b.visita || ""), b.nota || {});
    if (b.acao === "notaExcluir") return notaExcluir_(u, String(b.visita || ""), String(b.nota || ""));
    return { ok: false, erro: "ação desconhecida" };
  } finally { lock.releaseLock(); }
}
const doTecnico_ = (v, u) => v.resp && String(v.resp.login || "").toLowerCase() === u.login;
function campoListar_(u) {
  const nv = notasPorVisita_(), limite = Utilities.formatDate(new Date(Date.now() - 45 * 864e5), TZ, "yyyy-MM-dd");
  const visitas = ler_("visitas").filter(l => String(l.status) !== "Histórico").map(visita_)
    .filter(v => doTecnico_(v, u) && (v.status === "Em campo" || String(v.fim || "") >= limite))
    .map(v => ({ id: v.id, status: v.status, pedido: v.pedido, idObra: v.idObra, obra: v.obra, tipo: v.tipo, ini: v.ini, fim: v.fim,
      notas: (nv[v.id] || []).filter(n => n.login === u.login) }))
    .sort((a, b) => String(b.ini).localeCompare(String(a.ini)));
  return { ok: true, usuario: u, visitas: visitas };
}
function visitaAberta_(u, id) {
  const l = ler_("visitas").find(x => String(x.id) === id);
  if (!l) return { erro: "Visita não encontrada." };
  const v = visita_(l);
  if (!doTecnico_(v, u)) return { erro: "Você não é o responsável em campo desta visita." };
  if (v.status !== "Em campo") return { erro: "Esta visita já foi encerrada pelo supervisor." };
  return { v: v };
}
function notaSalvar_(u, idVisita, n) {
  const a = visitaAberta_(u, idVisita);
  if (a.erro) return { ok: false, erro: a.erro };
  const valor = r2_(n.valor), desc = limpa_(n.desc, 300);
  if (!(valor > 0)) return { ok: false, erro: "Informe o valor da nota." };
  if (!desc) return { ok: false, erro: "Diga o que foi comprado." };
  const id = "N" + Utilities.getUuid().replace(/-/g, "").slice(0, 12);
  let fotoId = "", link = "";
  const m = String(n.foto || "").match(/^data:image\/(jpeg|png|webp);base64,(.+)$/);
  if (m) {
    const blob = Utilities.newBlob(Utilities.base64Decode(m[2]), "image/" + m[1],
      ["Nota", a.v.idObra, a.v.pedido, limpa_(n.data, 10) || "", u.login, id].join(" ") + (m[1] === "png" ? ".png" : ".jpg"));
    const f = pasta_(PASTA_NOTAS).createFile(blob);
    fotoId = f.getId(); link = f.getUrl();
  }
  const mini = /^data:image\/jpeg;base64,/.test(String(n.mini || "")) && String(n.mini).length < 40000 ? String(n.mini) : "";
  gravar_("notas", { id: id, visita: idVisita, obra: limpa_(a.v.idObra + " - " + a.v.obra, 180), data: ehData_(n.data) ? n.data : Utilities.formatDate(new Date(), TZ, "yyyy-MM-dd"),
    valor: String(valor).replace(".", ","), descricao: desc, estabelecimento: limpa_(n.loja, 120), cnpj: limpa_(n.cnpj, 20), tecnico: u.nome, login: u.login,
    enviada_em: agora_(), recusada: "", foto_id: fotoId, foto_link: link, miniatura: mini, leitura: limpa_(n.leitura, 1500), digitado: n.digitado ? "Sim" : "" }, 0);
  atualizarTotais_(idVisita);
  return { ok: true, id: id };
}
function notaExcluir_(u, idVisita, idNota) {
  const a = visitaAberta_(u, idVisita);
  if (a.erro) return { ok: false, erro: a.erro };
  const n = ler_("notas").find(x => String(x.id) === idNota && String(x.visita) === idVisita);
  if (!n) return { ok: true };
  if (String(n.login) !== u.login) return { ok: false, erro: "Esta nota não é sua." };
  if (/^s/i.test(String(n.recusada || ""))) return { ok: false, erro: "Nota já analisada pelo supervisor." };
  try { if (n.foto_id) DriveApp.getFileById(String(n.foto_id)).setTrashed(true); } catch (e) { }
  aba_("notas").deleteRow(n._linha);
  atualizarTotais_(idVisita);
  return { ok: true };
}
// foto completa da nota: o supervisor vê todas; o técnico só as dele
function notaFoto_(idNota, u) {
  const n = ler_("notas").find(x => String(x.id) === idNota);
  if (!n || !n.foto_id) return { ok: false, erro: "Foto não encontrada." };
  if (u && String(n.login) !== u.login) return { ok: false, erro: "Esta nota não é sua." };
  const b = DriveApp.getFileById(String(n.foto_id)).getBlob();
  return { ok: true, foto: "data:" + b.getContentType() + ";base64," + Utilities.base64Encode(b.getBytes()) };
}
// mantém orçado, realizado e saldo da aba Visitas em dia quando entra ou sai uma nota
function atualizarTotais_(idVisita) {
  const l = ler_("visitas").find(x => String(x.id) === idVisita);
  if (!l) return;
  const v = visita_(l); v.notas = notasPorVisita_()[idVisita] || [];
  const t = totais_(v), c = T.visitas.cols;
  aba_("visitas").getRange(l._linha, c.indexOf("orcado") + 1, 1, 3).setValues([[String(t.orc), String(t.real), String(t.saldo)]]);
}

/* ---------- cópia de segurança semanal ---------- */
function backupSemanal() {
  const ss = planilha_(), pasta = pasta_(PASTA_BACKUP);
  DriveApp.getFileById(ss.getId()).makeCopy("Backup " + NOME_PLANILHA + " " + Utilities.formatDate(new Date(), TZ, "yyyy-MM-dd HH'h'mm"), pasta);
  const arqs = []; const it = pasta.getFiles();
  while (it.hasNext()) arqs.push(it.next());
  arqs.sort((a, b) => b.getDateCreated() - a.getDateCreated()).slice(BACKUPS_MANTER).forEach(f => f.setTrashed(true));
  return { ok: true };
}
function garantirGatilhoBackup_() {
  if (prop_("GATILHO_BACKUP")) return;
  if (!ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === "backupSemanal"))
    ScriptApp.newTrigger("backupSemanal").timeBased().onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(3).create();
  PropertiesService.getScriptProperties().setProperty("GATILHO_BACKUP", "1");
}

      return { id: "h" + v.id, status: "Histórico", origem: "Planilha BD - CUSTOS DE INSTALAÇÃO", tipo: v.tipo, pedido: v.pedido, id_obra: v.idObra, obra: v.obra,

}
